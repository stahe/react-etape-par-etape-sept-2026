    // =========================================================================
// 01-premiere-application/src/App.tsx - un premier composant
// -------------------------------------------------------------------------
// Un composant React est une FONCTION, dont le nom commence par une
// majuscule, qui RENVOIE ce qu'il faut afficher, écrit en JSX : du HTML
// dans le code TypeScript. React appelle cette fonction pour fabriquer la
// page ; il la RAPPELLE chaque fois que l'état du composant change, et met
// à jour la page avec le nouveau résultat.
// =========================================================================

import { useState } from 'react';
// une feuille de style, importée par le composant qui l'utilise
import './App.css';

// des données "ordinaires", hors du composant : elles ne changent jamais
const titre = 'Bienvenue dans React';
const langages = ['TypeScript', 'HTML', 'CSS'];

export default function App() {
  // useState(0) : une donnée d'ÉTAT, de valeur initiale 0. On reçoit sa
  // valeur ACTUELLE (compteur) et la fonction qui la change (setCompteur).
  // Appeler setCompteur demande à React de RAPPELER App pour mettre à jour
  // la page. On ne modifie jamais la page soi-même.
  const [compteur, setCompteur] = useState(0);
  const [heure, setHeure] = useState(() => new Date().toLocaleTimeString());

  // une fonction appelée par le bouton
  function incrementer() {
    setCompteur((n) => n + 1);
    setHeure(new Date().toLocaleTimeString());
  }

  // le JSX renvoyé : { expression } insère la valeur d'une expression ;
  // className remplace l'attribut HTML class (un mot réservé en JavaScript)
  return (
    <div className="container py-4">
      <h1 className="h3">{titre}</h1>
      <p>Cette page est fabriquée dans le navigateur, par React, à partir de index.html, main.tsx et du composant App.</p>
      <p>
        On y écrit du {langages.join(', ')} ({langages.length} langages).
      </p>

      {/* onClick : la fonction à appeler quand on clique (cf. exemple 04) */}
      <button className="btn btn-primary" onClick={incrementer}>
        Cliquez-moi
      </button>
      <p className="mt-3 compteur">
        Vous avez cliqué {compteur} fois (dernier clic : {heure}).
      </p>
    </div>
  );
}
    • ligne 11 : le hook [useState], fourni par [React] ;
    • ligne 13 : la feuille de style du composant ;
    • lignes 16-17 : deux données ordinaires, déclarées hors du composant : elles ne changent jamais ;
    • ligne 19 : le composant : une fonction dont le nom commence par une majuscule (c'est ainsi que le JSX distingue un composant, [<App />], d'un élément HTML, [<div>]). [export default] : c'est ce que le module exporte par défaut, importé dans [main.tsx] par [import App from './App'] ;
    • ligne 24 : [useState(0)] crée une donnée d'état, de valeur initiale 0. Il renvoie un tableau de deux éléments, que l'on « déstructure » : la valeur actuelle ([compteur]) et la fonction qui la change ([setCompteur]). L'état est mémorisé par [React] : il survit d'un appel de la fonction à l'autre ;
    • ligne 25 : on peut donner à [useState] une fonction qui calcule la valeur initiale : elle n'est appelée qu'au premier affichage ;
    • lignes 28-30 : une fonction appelée par le bouton. Elle ne touche pas à la page : elle change l'état. [setCompteur((n) => n + 1)] calcule la nouvelle valeur à partir de l'ancienne. Chaque appel d'une fonction [setXxx] demande à [React] de rappeler [App] ; [React] compare le nouveau résultat à l'ancien et ne modifie, dans la page, que ce qui a changé ;
    • lignes 35-51 : le JSX renvoyé, c'est-à-dire ce qu'il faut afficher :
      • le JSX ressemble au HTML, avec quelques différences : l'attribut [class] s'écrit [className] ([class] est un mot réservé de [JavaScript]), chaque balise doit être fermée ([<br />], [<input />]), et le tout doit avoir UN élément racine ;
      • ligne 37 : [{ expression }] insère la valeur d'une expression [TypeScript]. Elle est échappée : si [titre] contenait [<b>], on verrait [<b>] s'afficher, sans que la balise soit interprétée ;
      • ligne 40 : l'expression peut être quelconque : un appel de méthode, un calcul ;
      • ligne 43 : un commentaire, en JSX, s'écrit [{/* ... */}] ;
      • ligne 44 : un clic sur le bouton appelle la fonction [incrementer] (cf. exemple 04). On donne la fonction elle-même, sans parenthèses : [onClick={incrementer()}] l'appellerait tout de suite, pendant l'affichage ;
      • ligne 48 : la valeur actuelle de l'état. Quand l'état change, [App] est rappelée, et ce texte change.

    Le JSX n'est pas du HTML : c'est une écriture commode pour des appels de fonction. Le compilateur transforme [<h1 className="h3">{titre}</h1>] en un appel du genre [jsx('h1', { className: 'h3', children: titre })], qui fabrique un objet décrivant l'élément. Le composant renvoie donc un arbre d'objets (le « DOM virtuel »), que [React] compare à l'arbre précédent pour savoir quoi changer dans la vraie page.

// 02-etat/src/App.tsx - l'état, les valeurs calculées, useEffect
// -------------------------------------------------------------------------
// Un petit calculateur de prix TTC :
//   - useState    : une valeur d'ÉTAT ; la changer (setXxx) fait rappeler
//                   le composant, qui fabrique la page avec la nouvelle valeur ;
//   - une valeur CALCULÉE à partir de l'état n'est PAS un état : c'est une
//     simple variable, recalculée à chaque appel du composant ;
//   - useMemo     : la même chose, mais le calcul n'est refait que si ses
//                   dépendances ont changé (utile pour un calcul coûteux) ;
//   - useEffect   : du code exécuté APRÈS l'affichage, quand certaines
//                   valeurs ont changé : un "effet de bord" (agir sur autre
//                   chose que la page : ici, le titre de l'onglet).
// =========================================================================

import { useEffect, useMemo, useState } from 'react';

// un formateur de montants en euros, à la française (1 234,50 €)
const euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

export default function App() {
  const [libelle, setLibelle] = useState('Stylo');
  const [prixHT, setPrixHT] = useState(2.5);
  const [quantite, setQuantite] = useState(1);

  // nouveau, on ajuste l'état PENDANT l'appel du composant (React
  // recommence alors aussitôt l'appel, avant d'afficher quoi que ce soit)
  const [historique, setHistorique] = useState<{ id: number; texte: string }[]>([]);
  const [ancienTTC, setAncienTTC] = useState(totalTTC);
  if (totalTTC !== ancienTTC) {
    setAncienTTC(totalTTC);

import { useRef, useState, type FormEvent, type KeyboardEvent, type MouseEvent } from 'react';

interface Tache {
  id: number;
  libelle: string;
  faite: boolean;
}

// le style de la zone cliquable : un objet, comme pour l'attribut style

  const [taches, setTaches] = useState<Tache[]>([
    { id: 1, libelle: 'Installer Node.js', faite: true },
    { id: 2, libelle: 'Lire le chapitre sur les événements', faite: false },
  ]);
  const [journal, setJournal] = useState<string[]>([]);
  // une référence au <input> de la nouvelle tâche (null tant qu'il n'est pas affiché)
  const champ = useRef<HTMLInputElement>(null);
  // une référence peut aussi garder une valeur qui n'est PAS affichée : la
  // modifier ne rappelle pas le composant
  const prochainId = useRef(3);

  function tracer(message: string) {
    setJournal((j) => [`${new Date().toLocaleTimeString()} ${message}`, ...j]);
  }

  // onSubmit : Entrée dans le champ OU clic sur le bouton. preventDefault()
  // empêche le navigateur d'envoyer le formulaire (et de recharger la page)
  function ajouter(evenement: FormEvent<HTMLFormElement>) {
    evenement.preventDefault();
    const input = champ.current!;
    const libelle = input.value.trim();
    if (libelle === '') {
      return;
    }
    // un état qui contient un tableau : on le REMPLACE par un nouveau tableau
    // (on ne modifie jamais l'ancien : React ne verrait pas le changement)
    setTaches((t) => [...t, { id: prochainId.current++, libelle, faite: false }]);
    tracer(`ajout de « ${libelle} »`);
    input.value = '';
    input.focus();
  }

    if (evenement.key === 'Escape') {
      evenement.currentTarget.value = '';

  }

    setTaches((t) => t.map((x) => (x.id === tache.id ? { ...x, faite } : x)));
    tracer(`« ${tache.libelle} » : ${faite ? 'faite' : 'à faire'}`);
  }

  function supprimer(tache: Tache) {
    setTaches((t) => t.filter((x) => x.id !== tache.id));
    tracer(`suppression de « ${tache.libelle} »`);
  }

  // l'événement d'origine (nativeEvent) a les coordonnées relatives à la zone

    tracer(`clic en (${evenement.nativeEvent.offsetX}, ${evenement.nativeEvent.offsetY})`);
  }

        <button type="submit" className="btn btn-primary">Ajouter</button>
      </form>

            <span className={`flex-grow-1 ${t.faite ? 'text-decoration-line-through text-muted' : ''}`}>{t.libelle}</span>

          </li>
        ))}
      </ul>

// dans l'événement dépend du champ :
//   - input texte, textarea          -> e.target.value (string)
//   - input type="number"            -> e.target.valueAsNumber (NaN si vide)
//   - case à cocher                  -> e.target.checked (boolean)
//   - boutons radio                  -> e.target.value du bouton choisi
//   - select                         -> e.target.value
//   - select multiple                -> e.target.selectedOptions (plusieurs)
//   - input type="date"              -> e.target.value, "yyyy-MM-dd"

// 06-use-reducer/src/App.tsx - un formulaire, un objet, un réducteur
// -------------------------------------------------------------------------
// L'exemple 05 avait un useState par champ. Pour un vrai formulaire, on
// préfère UN objet qui contient toutes les valeurs, le MODÈLE, et l'on
// décrit ses changements dans une seule fonction, le RÉDUCTEUR :
//
//     reducteur(etatActuel, action) -> nouvelEtat
//
// useReducer(reducteur, etatInitial) renvoie l'état et une fonction
// dispatch(action) : le composant ne modifie plus l'état, il DÉCRIT ce qui
// s'est passé ({ type: 'modifier', champ: 'nom', valeur: 'Dupont' }) et le
// réducteur en déduit le nouvel état. Toute la logique est en un endroit,
// et c'est une fonction PURE (même entrée, même sortie, aucun effet de
// bord), facile à tester.
//
// L'état garde aussi, pour chaque champ, s'il a été "touché" (quitté au
// moins une fois) ; on en déduit s'il est "modifié" (sa valeur diffère de
// la valeur de départ) et s'il est désactivé.
// =========================================================================

import { useReducer, type ChangeEvent } from 'react';

// la forme des données du formulaire
interface Profil {
  titre: string;
  prenom: string;
  nom: string;
  age: number;
  newsletter: boolean;
  email: string;
  pays: string;
  naissance: string;
  presentation: string;
}

const DEPART: Profil = {
  titre: 'Mme',
  prenom: 'Marie',
  nom: 'PELISSIER',
  age: 30,
  newsletter: false,
  email: '',
  pays: 'fr',
  naissance: '',
  presentation: '',
};

type Champ = keyof Profil; // 'titre' | 'prenom' | ... : le nom d'un champ

interface Etat {
  valeurs: Profil;
  touches: Partial<Record<Champ, boolean>>; // les champs déjà quittés
}

// les actions possibles : une "union discriminée" par la propriété type
type Action =
  | { type: 'modifier'; champ: Champ; valeur: Profil[Champ] }
  | { type: 'quitter'; champ: Champ }
  | { type: 'majuscules' }
  | { type: 'reinitialiser' };

// le réducteur : il ne MODIFIE jamais l'état reçu, il en renvoie un nouveau
function reducteur(etat: Etat, action: Action): Etat {
  switch (action.type) {
    case 'modifier':
      return { ...etat, valeurs: { ...etat.valeurs, [action.champ]: action.valeur } };
    case 'quitter':
      return { ...etat, touches: { ...etat.touches, [action.champ]: true } };
    case 'majuscules': {
      const { prenom, nom } = etat.valeurs;
      return { ...etat, valeurs: { ...etat.valeurs, prenom: prenom.toUpperCase(), nom: nom.toUpperCase() } };
    }
    case 'reinitialiser':
      return { valeurs: { ...DEPART }, touches: {} };
  }
}

// l'appelle à chaque affichage, les erreurs sont donc toujours à jour.

//

// réponse immédiate, sans aller-retour avec le serveur). Elle ne protège

import { useState, type FormEvent } from 'react';

interface Compte {
  titre: string;
  prenom: string;
  nom: string;
  login: string;
  password: string;
  confirmation: string;
}

type Champ = keyof Compte;
// au plus un message d'erreur par champ
type Erreurs = Partial<Record<Champ, string>>;

const TITRES = ['Mr', 'Mme', 'Melle'];
const FORMAT_LOGIN = /^[a-z0-9._-]{1,30}$/;
const MOT_DE_PASSE_MIN = 6;

// les règles : une fonction pure, qui ne dépend que de ses arguments
function valider(c: Compte): Erreurs {
  const erreurs: Erreurs = {};
  if (!TITRES.includes(c.titre)) erreurs.titre = 'Choisissez un titre dans la liste.';
  if (c.prenom.trim() === '') erreurs.prenom = 'Le prénom est obligatoire.';
  else if (c.prenom.length > 20) erreurs.prenom = 'Le prénom ne doit pas dépasser 20 caractères.';
  if (c.nom.trim() === '') erreurs.nom = 'Le nom est obligatoire.';
  else if (c.nom.length > 20) erreurs.nom = 'Le nom ne doit pas dépasser 20 caractères.';
  if (c.login === '') erreurs.login = 'Le login est obligatoire.';
  else if (!FORMAT_LOGIN.test(c.login)) erreurs.login = 'Lettres minuscules non accentuées, chiffres, point, tiret, souligné.';
  if (c.password === '') erreurs.password = 'Le mot de passe est obligatoire.';
  else if (c.password.length < MOT_DE_PASSE_MIN) erreurs.password = `Au moins ${MOT_DE_PASSE_MIN} caractères.`;
  // une règle qui compare DEUX champs
  if (c.confirmation !== c.password) erreurs.confirmation = 'La confirmation ne correspond pas au mot de passe.';
  return erreurs;
}

const VIDE: Compte = { titre: 'Mr', prenom: '', nom: '', login: '', password: '', confirmation: '' };
const TOUS: Record<Champ, boolean> = { titre: true, prenom: true, nom: true, login: true, password: true, confirmation: true };

export default function App() {
  const [compte, setCompte] = useState<Compte>(VIDE);
  const [touches, setTouches] = useState<Partial<Record<Champ, boolean>>>({});
  const [envoiTente, setEnvoiTente] = useState(false);
  const [succes, setSucces] = useState(false);

  // les erreurs : recalculées à chaque affichage, jamais rangées dans l'état
  const erreurs = valider(compte);
  const nbErreurs = Object.keys(erreurs).length;

  // la classe Bootstrap d'un champ : rouge (is-invalid) ou vert (is-valid),
  // seulement une fois qu'il a été touché
  function classe(champ: Champ, base = 'form-control'): string {
    if (!touches[champ]) return base;
    return `${base} ${erreurs[champ] ? 'is-invalid' : 'is-valid'}`;
  }

  // les attributs communs à tous les champs
  function lier(champ: Champ) {
    return {
      id: champ,
      value: compte[champ],
      onChange: (e: { target: { value: string } }) => {
        setCompte((c) => ({ ...c, [champ]: e.target.value }));
        setSucces(false);
      },
      onBlur: () => setTouches((t) => ({ ...t, [champ]: true })),
    };
  }

  function envoyer(evenement: FormEvent<HTMLFormElement>) {
    evenement.preventDefault();
    setEnvoiTente(true);
    setTouches(TOUS); // toutes les erreurs apparaissent
    // ici, on enverrait le compte au serveur (cf. exemple 25)
    setSucces(nbErreurs === 0);
  }

  return (
    <div className="container py-4" style={{ maxWidth: 640 }}>

          <label className="form-label" htmlFor="titre">Titre</label>

            {TITRES.map((t) => <option key={t}>{t}</option>)}
          </select>
          <div className="invalid-feedback">{erreurs.titre}</div>
        </div>
        <div className="row g-3 mb-3">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="prenom">Prénom</label>
            {/* is-invalid encadre le champ de rouge ET fait apparaître le
                .invalid-feedback qui le suit : on y écrit l'erreur du champ */}
            <input className={classe('prenom')} {...lier('prenom')} />
            <div className="invalid-feedback">{erreurs.prenom}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="nom">Nom</label>
            <input className={classe('nom')} {...lier('nom')} />
            <div className="invalid-feedback">{erreurs.nom}</div>
          </div>
        </div>
        <div className="mb-3">
          <label className="form-label" htmlFor="login">Login</label>
          <input className={classe('login')} autoComplete="username" {...lier('login')} />
          <div className="invalid-feedback">{erreurs.login}</div>
        </div>
        <div className="row g-3 mb-4">
          <div className="col-sm-6">
            <label className="form-label" htmlFor="password">Mot de passe</label>
            <input type="password" className={classe('password')} autoComplete="new-password" {...lier('password')} />
            <div className="invalid-feedback">{erreurs.password}</div>
          </div>
          <div className="col-sm-6">
            <label className="form-label" htmlFor="confirmation">Confirmation</label>
            <input type="password" className={classe('confirmation')} autoComplete="new-password" {...lier('confirmation')} />
            <div className="invalid-feedback">{erreurs.confirmation}</div>
          </div>
        </div>

        )}
        {succes && <div className="alert alert-success mt-3 mb-0">Formulaire valide : il pourrait être envoyé au serveur.</div>}
      </form>
    </div>
  );
}

  mfin: number;

// "8:05" plutôt que "8:5"
function heure(h: number, m: number): string {
  return `${h}:${String(m).padStart(2, '0')}`;
}

export function horaire(c: Creneau, partie: 'tout' | 'debut' | 'fin' = 'tout'): string {
  switch (partie) {
    case 'debut':
      return heure(c.hdebut, c.mdebut);
    case 'fin':
      return heure(c.hfin, c.mfin);
    default:
      return `${heure(c.hdebut, c.mdebut)} - ${heure(c.hfin, c.mfin)}`;
  }
}

// "marie pelissier" -> "Marie Pelissier"
export function majusculeInitiale(texte: string): string {
  return texte.replace(/\p{L}+/gu, (mot) => mot[0].toUpperCase() + mot.slice(1).toLowerCase());
}

// "Marie PELISSIER" -> "MP"
export function initiales(nomComplet: string): string {
  return nomComplet
    .split(/\s+/)
    .filter((mot) => mot !== '')
    .map((mot) => mot[0].toUpperCase())
    .join('');
}

// "2026-10-05" (le format de l'API RdvMedecins) -> une Date, à minuit UTC
export function versDate(jour: string): Date {
  return new Date(`${jour}T00:00:00Z`);
}

    • lignes 25-34 : « 8:00 - 8:20 », ou seulement le début, ou la fin, selon le paramètre [partie], facultatif (valeur par défaut ['tout']) ;

    • lignes 51-53 : une date de l'API (["2026-10-05"]) devient un objet [Date], à minuit UTC.
